## 1. หลักการสำคัญ

## 5. โค้ดตั้งต้น (rule engine)

```python
import numpy as np
import pandas as pd

TOL = 0.03  # ใช้กับค่าเดี่ยวเท่านั้น
INF = float('inf')

def pt(v):            # ค่าเดี่ยว -> ช่วง +/- TOL
    return (v - TOL, v + TOL)

def ge(v): return (v, INF)       # ขึ้นไป
def le(v): return (-INF, v)      # ลงมา

# แต่ละกฎ: (id, ah_list, fav, draw, dog)  None = ไม่เช็ก
RULES = [
    ('R1', [-0.25], le(2.09), ge(3.25), ge(2.80)),
    ('R1', [-0.5],  le(1.85), ge(3.30), ge(4.00)),
    ('R2a', [-0.25], ge(2.20), le(3.25), le(2.99)),
    ('R2a', [-0.5],  ge(1.83), le(3.60), le(3.80)),
    ('R2b', [-0.5],  ge(1.92), le(3.55), le(3.75)),
    ('R2c', [-0.5],  ge(2.00), le(3.50), le(3.65)),
    ('R3a', [0, -0.25], None, ge(3.65), None),
    ('R3b', [-1.5],  None, ge(4.40), ge(7.20)),   # + ห้ามไหลลง (ดูหัวข้อ 6)
    ('R4a', [0, -0.25], None, ge(3.65), None),    # + เงื่อนไข Ladbrokes (หัวข้อ 6)
    ('R4b', [-0.5],  None, ge(3.70), None),
    ('R5_1', [-0.25], pt(2.20), pt(3.50), pt(2.99)),
    ('R5_2', [-0.5],  pt(1.99), pt(3.50), pt(3.50)),
    ('R5_3', [-0.75], (1.80, 1.89), pt(3.80), pt(4.40)),
    ('R5_4', [-0.75], (1.72, 1.79), pt(3.60), pt(3.90)),
    ('R5_5', [-1.0],  (1.52, 1.65), pt(4.50), pt(5.00)),
    ('R6_1', [-0.25], (2.00, 2.09), pt(3.25), pt(3.10)),
]

def _dist(x, rng):
    if rng is None:
        return 0.0
    lo, hi = rng
    return 0.0 if lo <= x <= hi else min(abs(x - lo), abs(x - hi))

def rule_features(row):
    out = {}
    for rid, ahs, fav, draw, dog in RULES:
        key = f'{rid}_ah{ahs[0]}'
        if not any(np.isclose(row.ah_open, a) for a in ahs):
            out[key + '_match'] = 0
            out[key + '_near'] = 0
            out[key + '_dist'] = np.nan
            continue
        d = max(_dist(row.p_fav, fav), _dist(row.p_draw, draw), _dist(row.p_dog, dog))
        out[key + '_match'] = int(d == 0)
        out[key + '_near'] = int(d <= TOL)
        out[key + '_dist'] = d
    return pd.Series(out)

```

## 6. เงื่อนไขที่ต้องมีข้อมูลเพิ่ม (ทำเป็น feature แยก)

1. **ห้ามไหลลง (R3b):** ต้องมีราคาเปิดและราคาปิด สร้าง `fav_no_drop = int(p_fav_close >= p_fav_open)` และ `dog_no_drop` ถ้าใช้ราคาปิด ต้องกำหนดเวลาทำนายให้ชัดว่าเป็นก่อนเตะกี่นาที ห้ามใช้ราคาที่เกิดหลังเวลาทำนาย (data leakage)
2. **Ladbrokes (R4a):** ต้องมีราคา Over HT ของ Ladbrokes ตอนเปิดและตอนปิด ถ้า API ไม่มีเจ้านี้ ให้ตั้ง feature เป็น NaN แล้วรายงานว่าไม่สามารถทดสอบกฎนี้ได้ อย่าแทนด้วยเจ้ามืออื่นโดยไม่บอก
3. **รูปแบบราคาน้ำ:** "0.50-0.75" น่าจะเป็นราคาแบบ Malay/HK ถ้า API คืนราคา decimal ให้แปลง (HK = decimal - 1) แล้วบันทึกวิธีแปลงในรายงาน

## 7. วิธีเทรน

1. สร้าง feature จากกฎทุกข้อ + feature เดิมใน Brief (หัวข้อ 3) เก็บทุกอย่างเป็นคอลัมน์ ห้ามกรองนัดทิ้งด้วยกฎ
2. เริ่มจาก Logistic Regression (multi-output: `p_fav_cover`, `p_over`) แล้วค่อยลอง XGBoost
3. แบ่ง train/test ตามเวลา เก่า→ใหม่ walk-forward ห้ามสุ่ม
4. หลังเทรน รายงานความสำคัญของแต่ละกฎ และ **อัตราถูกของแต่ละกฎแยกตามตลาด** (AH หรือ O/U) พร้อมจำนวนนัดที่เข้ากฎ (n) และช่วงความคลาดเคลื่อน ถ้า n น้อยกว่า 30 ให้เขียนกำกับว่าสรุปไม่ได้
5. เทียบกับ baseline จากราคาน้ำที่ตัดค่าน้ำแล้ว ตามเกณฑ์ผ่านใน Brief หัวข้อ 7
6. ผลใกล้เคียงสุ่มเป็นผลที่ยอมรับได้ ต้องไม่ปรับกฎเพื่อให้ผลดีขึ้น

## 8. จุดที่ต้องให้ผู้ใช้ยืนยันก่อนเทรน

- ลีกและเจ้ามือที่ใช้ (กฎ R4a ระบุ Ladbrokes)

## 9. ตรวจรูปแบบราคา (ฮ่องกง / ทศนิยม / มาเลย์ ฯลฯ) ก่อนใช้กฎ

กฎทุกข้อมีค่าตัวเลขเป็นเกณฑ์ ถ้าราคาที่ดึงจาก API อยู่คนละรูปแบบกับราคาในภาพ เกณฑ์จะผิดทั้งหมดโดยไม่มี error เตือน ให้ตรวจและแปลงก่อนสร้าง feature

### 9.1 สิ่งที่อนุมานได้จากตัวเลขในภาพ (ยังต้องให้ผู้ใช้ยืนยัน)

| ราคา | รูปแบบที่น่าจะเป็น | เหตุผล |
| --- | --- | --- |
| ราคาพูล 1X2 (2.09, 3.25, 2.80 ฯลฯ) | **ทศนิยม (decimal / ยุโรป)** | ค่าอยู่ในช่วง 1.5-5.0 และราคาเสมอสูงกว่าราคาทีมต่อ ตรงรูปแบบ 1X2 ทศนิยม |
| ค่าน้ำสกอร์สูงของ Ladbrokes (0.50-0.75) | **น่าจะเป็นแบบฮ่องกง (HK) หรือมาเลย์** | ต่ำกว่า 1 ได้ ซึ่งเกิดได้เฉพาะรูปแบบที่ไม่รวมเงินต้น แต่ Ladbrokes ปกติแสดง decimal/fractional ต้องตรวจว่าตัวเลขนี้ผู้ใช้ดูจากหน้าเว็บไหน ตั้งค่าแสดงผลแบบใด |

### 9.2 ตารางแปลงรูปแบบราคา

| จาก | เป็น decimal |
| --- | --- |
| ฮ่องกง (HK) | decimal = HK + 1 (เช่น 0.75 -> 1.75) |
| มาเลย์ (บวก) | decimal = Malay + 1 |
| มาเลย์ (ลบ) | decimal = 1 + 1 / abs(Malay) |
| เศษส่วน (UK) a/b | decimal = a/b + 1 (เช่น 3/4 -> 1.75) |
| อินโดนีเซีย (บวก) | decimal = Indo + 1 |

ดังนั้นเงื่อนไข Ladbrokes ในกฎ R4a ที่ว่า "ค่าน้ำเปิด 0.50 แต่ไม่เกิน 0.75" ถ้าเป็น HK จะเท่ากับ decimal **1.50 ถึง 1.75** ถ้าเป็น decimal อยู่แล้วช่วง 0.50-0.75 จะเป็นไปไม่ได้ (decimal ต้องมากกว่า 1) ให้เก็บทั้งค่าที่ตีความและเหตุผลไว้ในรายงาน

### 9.3 วิธีตรวจจากข้อมูลจริงที่ดึงมา (ทำเป็นโค้ดทดสอบ)

```python
def overround(p_fav, p_draw, p_dog):
    return 1/p_fav + 1/p_draw + 1/p_dog

def detect_format(price):
    # ราคาต่ำกว่า 1 = รูปแบบที่ไม่รวมเงินต้น (HK / มาเลย์ / อินโด) ไม่ใช่ decimal
    if price < 1.0:
        return 'non_decimal (HK/Malay/Indo?)'
    return 'decimal (or fractional converted)'

# 1X2 แบบ decimal: overround ควรอยู่ราว 1.02-1.15
# ถ้าได้ค่าต่ำกว่า 1.0 หรือสูงกว่า 1.3 แสดงว่ารูปแบบหรือการอ่านคอลัมน์ผิด
```

### 9.4 ข้อสังเกตสำคัญจากการคำนวณ overround ของราคาในภาพ

ถ้าอ่านภาพเป็น decimal และคอลัมน์เป็น ต่อ/เสมอ/รอง ค่า overround ของแถวตัวอย่างอยู่ราว **1.07-1.14** (เช่น R1 AH -0.25: 1/2.09 + 1/3.25 + 1/2.80 ≈ 1.14, R5\_1: ≈ 1.08) ซึ่งสูงกว่าเจ้ามือใหญ่อย่าง Pinnacle (ปกติราว 1.02-1.04) หมายความว่า

- ราคาในภาพน่าจะมาจากเจ้ามือ/เว็บที่ค่าหักสูงกว่า หรือเป็นราคาคนละประเภท ไม่ใช่ราคาของ Pinnacle ที่ Brief แนะนำ
- **ห้ามเอาเกณฑ์ในภาพไปใช้กับราคา Pinnacle ตรง ๆ** ถ้าเจ้ามือคนละเจ้า ราคา 1X2 จะเลื่อนไปคนละระดับและกฎจะไม่ตรงกับนัดที่ผู้ทำนายตั้งใจ
- ทางเลือก: (ก) หาเจ้ามือเดียวกับที่ผู้ทำนายใช้ แล้วดึงราคาจากเจ้ามือนั้น หรือ (ข) แปลงเป็นความน่าจะเป็นที่ตัดค่าน้ำแล้ว (no-vig) ทั้งเกณฑ์และราคาที่ดึงมา แล้วเทียบกันที่ระดับความน่าจะเป็นแทนราคาดิบ

### 9.5 สิ่งที่ต้องบันทึกในฐานข้อมูลทุกแถวราคา

- `bookmaker` (ชื่อเจ้ามือ), `odds_format_raw` (รูปแบบที่ API ให้มา), `odds_format_stored` (ให้เก็บเป็น decimal เสมอ), `captured_at` (เวลาของราคา), `is_open_price` (0/1)
- ราคาที่แปลงแล้วต้องเก็บคู่กับราคาดิบเสมอ เพื่อตรวจย้อนหลังได้

### 9.6 คำถามที่ต้องให้ผู้ใช้ตอบ

1. ตัวเลขในภาพได้มาจากเว็บ/เจ้ามือไหน และตั้งแสดงราคาเป็นแบบใด (Decimal, HK, Malay)
2. ค่าน้ำ Ladbrokes 0.50-0.75 ผู้ใช้อ่านจากหน้าเว็บ Ladbrokes โดยตรง หรือจากเว็บรวมราคาที่แปลงเป็น HK ให้
3. ราคาพูล 1X2 ในภาพเป็นราคาของเจ้ามือเจ้าเดียวกันทั้งหมด หรือรวมหลายเจ้า
